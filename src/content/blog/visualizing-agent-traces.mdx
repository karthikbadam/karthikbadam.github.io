import { useState } from 'react';
import { Box, SimpleGrid, Stat, Heading, Text } from '@chakra-ui/react';
import { useTraceData } from '../../hooks/useTraceData';
import { StackedBarChart } from '../../components/StackedBarChart';
import { SpanGanttChart } from '../../components/SpanGanttChart';
import { TraceIcicle } from '../../components/TraceIcicle';
import { SpanInspector } from '../../components/SpanInspector';

export function TraceVisualization() {
  const { data, loading, error } = useTraceData();
  const [selectedSpan, setSelectedSpan] = useState(null);

  if (loading) return <Text>Loading trace data...</Text>;
  if (error) return <Text color="red.500">Error: {error}</Text>;
  if (!data) return null;

  return (
    <>
      {/* Basic Statistics */}
      <Box mb={8}>
        <Heading size="lg" mb={4}>Trace Metrics</Heading>
        <SimpleGrid columns={{ base: 2, md: 3 }} gap={4}>
          <Stat.Root>
            <Stat.Label>Elapsed Time</Stat.Label>
            <Stat.ValueText>{data.metrics.total_duration.toFixed(2)}s</Stat.ValueText>
            <Stat.HelpText>Actual trace runtime</Stat.HelpText>
          </Stat.Root>
          <Stat.Root>
            <Stat.Label>Total Spans</Stat.Label>
            <Stat.ValueText>{data.metrics.total_spans}</Stat.ValueText>
          </Stat.Root>
          <Stat.Root>
            <Stat.Label>LLM Calls</Stat.Label>
            <Stat.ValueText>{data.metrics.llm_call_count}</Stat.ValueText>
          </Stat.Root>
          <Stat.Root>
            <Stat.Label>Total Tokens</Stat.Label>
            <Stat.ValueText>{data.metrics.total_tokens.toLocaleString()}</Stat.ValueText>
          </Stat.Root>
          <Stat.Root>
            <Stat.Label>Avg Step Duration</Stat.Label>
            <Stat.ValueText>{data.metrics.avg_step_duration.toFixed(2)}s</Stat.ValueText>
          </Stat.Root>
          <Stat.Root>
            <Stat.Label>Max Nesting Depth</Stat.Label>
            <Stat.ValueText>{data.metrics.max_nesting_depth}</Stat.ValueText>
          </Stat.Root>
        </SimpleGrid>
      </Box>

      {/* Aggregated Bar Chart */}
      <Box mb={8}>
        <Heading size="lg" mb={4}>Activity Over Time (Aggregated)</Heading>
        <Text mb={4}>
          This stacked bar chart shows how different span types contribute to the overall execution time, grouped into 10-second buckets.
        </Text>
        <StackedBarChart data={data.buckets.aggregated} />
      </Box>

      {/* Gantt Chart for Individual Spans */}
      <Box mb={8}>
        <Heading size="lg" mb={4}>Detailed Span Timeline (Gantt View)</Heading>
        <Text mb={4}>
          This Gantt chart shows each span's exact timing and duration. Click on any span to inspect its details.
        </Text>
        <SpanGanttChart 
          data={data.buckets.individual} 
          onSpanClick={(span) => setSelectedSpan(span)}
        />
      </Box>

      {/* Icicle Chart */}
      <Box mb={8}>
        <Heading size="lg" mb={4}>Hierarchical Trace View</Heading>
        <Text mb={4}>
          This icicle chart reveals the 7-layer hierarchy of the trace, from execution phases down to individual ReAct components.
        </Text>
        <TraceIcicle 
          hierarchyData={data.icicle}
          onNodeClick={(node) => setSelectedSpan(node)}
        />
      </Box>

      {/* Span Inspector */}
      <SpanInspector 
        selectedSpan={selectedSpan}
        onClose={() => setSelectedSpan(null)}
      />
    </>
  );
}
